SDK Configuration:
Point your Temporal SDK to the profiler instead of the Temporal server:

  Go:
    client.Dial(client.Options{
        HostPort: "{{ include "temporal-profiler.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.grpcPort }}",
    })

  TypeScript:
    await Connection.connect({
        address: '{{ include "temporal-profiler.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.grpcPort }}',
    });

  Python:
    await Client.connect("{{ include "temporal-profiler.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.grpcPort }}")
